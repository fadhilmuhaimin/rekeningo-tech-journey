# Pembanding: OFFSET 3 sesudah kiriman Dimas (cara M12) mengulang baris terakhir halaman pertama
=> SELECT p.jurnal_id, j.jenis, p.jumlah FROM posting p JOIN jurnal j ON j.id = p.jurnal_id
   WHERE p.buku_id = (SELECT id FROM buku WHERE akun_id = 419) ORDER BY p.dibuat DESC, p.id DESC LIMIT 3 OFFSET 3
