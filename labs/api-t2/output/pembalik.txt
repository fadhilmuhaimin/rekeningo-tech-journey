# --- B. Admin membalik jurnal itu dengan alasan; jurnal asal tidak diubah ---
$ curl -i -X POST '/jurnal/1130/pembalik' -H 'Authorization: Bearer <token sesi admin>' -d '{"alasan": "Budi salah memilih warung, uang dikembalikan atas konfirmasi Warung Sari"}'
HTTP/1.1 201 Created
Content-Type: application/json

{"id":1131,"membalik":1130}
=> SELECT j.id, j.jenis, j.membalik_jurnal_id, b.jenis AS buku, a.nama, p.jumlah FROM jurnal j JOIN posting p ON p.jurnal_id = j.id JOIN buku b ON b.id = p.buku_id JOIN akun a ON a.id = b.akun_id WHERE j.id IN (1130, 1131) ORDER BY p.id

# --- C. Yang ditolak: dibalik dua kali, alasan pendek, bukan admin, koreksi gaya Tahap 1 ---
