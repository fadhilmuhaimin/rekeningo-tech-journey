# --- C. Log berantai: aksi yang sama, lalu pemilik tabel (bukan superuser) mengubah, menambah, dan menghapus ---
=> SELECT id, pelaku_id, aksi, objek, penyetuju_id,
          hash_sebelumnya = coalesce((SELECT b.hash FROM audit_log b WHERE b.id = a.id - 1), repeat('0', 64)) AS bersambung
   FROM audit_log a ORDER BY id

----+-----------+------------------+--------------------------------+--------------+------------

 2  | 420       | lihat_verifikasi | verifikasi:1                   |              | t
 3  | 420       | buka_berkas      | verifikasi:1/kartu             |              | t
 4  | 420       | bekukan          | akun:40001                     |              | t
