# --- F. Contract: jumlah tidak dikirim dan tidak dibaca lagi, sesudah tidak ada app 1.0 ---
# Di lab: Budi memperbarui app ke 1.1, jadi tidak ada lagi HP yang membaca atau mengirim jumlah.
# Kode contract = kode expand dengan perubahan ini:
#   internal/handler/handler.go  Jumlah *int64 `json:"jumlah"`  ->  `json:"-"`   (request bayar)
#   internal/handler/handler.go  []string{"ke", "jumlah"}  ->  []string{"ke", "nominal"}
#   internal/service/bayar.go    ErrValidasi{"jumlah",  ->  ErrValidasi{"nominal",
#   internal/repo/transaksi.go   Jumlah int64 `json:"jumlah"`  ->  `json:"-"`   (GET /transfers/{id})
#   internal/repo/riwayat.go     Jumlah int64 `json:"jumlah"`  ->  `json:"-"`   (riwayat)
$ go build -o api-t1-contract ./cmd/api
