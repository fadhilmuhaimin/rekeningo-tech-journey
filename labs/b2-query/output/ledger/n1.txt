# --- A. Cara M11: daftar dulu, lalu nama lawan satu query per baris ---
=> SELECT p.id, p.jurnal_id, j.jenis, p.jumlah, p.dibuat
   FROM posting p JOIN jurnal j ON j.id = p.jurnal_id
   WHERE p.buku_id = %(buku)s
   ORDER BY p.dibuat DESC, p.id DESC LIMIT 20
   -- lalu untuk setiap baris:
=> SELECT coalesce(a.nama, CASE b.jenis WHEN 'tunjangan' THEN 'Tunjangan makan' WHEN 'kas_pencairan' THEN 'Pencairan' END)
   FROM posting q JOIN buku b ON b.id = q.buku_id LEFT JOIN akun a ON a.id = b.akun_id
   WHERE q.jurnal_id = %(jurnal)s AND q.buku_id <> %(buku)s
   ORDER BY q.id LIMIT 1

# --- B. ADR 15: satu query, lawan dicari di dalam query yang sama ---
=> SELECT p.id, p.jurnal_id, j.jenis, p.jumlah, p.dibuat,
          coalesce(la.nama, CASE lb.jenis WHEN 'tunjangan' THEN 'Tunjangan makan' WHEN 'kas_pencairan' THEN 'Pencairan' END) AS lawan
   FROM posting p
   JOIN jurnal j ON j.id = p.jurnal_id
   LEFT JOIN LATERAL (SELECT q.buku_id FROM posting q
                      WHERE q.jurnal_id = p.jurnal_id AND q.buku_id <> p.buku_id ORDER BY q.id LIMIT 1) l ON true
   LEFT JOIN buku lb ON lb.id = l.buku_id
   LEFT JOIN akun la ON la.id = lb.akun_id
   WHERE p.buku_id = %(buku)s
   ORDER BY p.dibuat DESC, p.id DESC LIMIT 20
