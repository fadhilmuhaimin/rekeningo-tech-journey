# --- A. Arsip sumber: dua partisi lama dilepas dan diekspor oleh kode lab t5-partisi ---
# Partisi posting_lalu_02 dan posting_lalu_01 dilepas dengan DETACH PARTITION ... CONCURRENTLY, lalu diekspor ke CSV tanpa enkripsi (lepas.ekspor).
# Tiap file CSV: 28.000 baris. Tidak ada enkripsi sampai langkah berikut.
$ curl -X POST $BAO/v1/sys/mounts/transit ; curl -X POST $BAO/v1/transit/keys/kek-arsip -d '{"type": "aes256-gcm96"}'   (operator)
# kunci kek-arsip: versi terbaru 1, min_decryption_version 1, min_encryption_version 0, min_available_version 0, versi yang tampil di daftar kunci [1]
# Tiga kebijakan, masing-masing satu jalur dengan satu hak (update): pengarsip = datakey/plaintext, pembuka = decrypt, pembungkus-ulang = rewrap.
# Tiap peran punya token sendiri. KEK tidak punya jalur baca: tidak ada perintah yang menjawab isi kunci.

# --- B. Arsipkan: satu data key per file, dibungkus KEK, dan dicatat di arsip_partisi dan jangkar ---
$ BAO_TOKEN=<token pengarsip> kmsarsip arsipkan -sumber posting_lalu_02.csv -tujuan posting_lalu_02.enc -periode <tanggal> -jangkar jangkar-arsip.jsonl

$ BAO_TOKEN=<token pengarsip> kmsarsip arsipkan -sumber posting_lalu_01.csv -tujuan posting_lalu_01.enc -periode <tanggal> -jangkar jangkar-arsip.jsonl

# Ukuran berkas terenkripsi = ukuran file asal + 28 byte (nonce 12, tag 16), dan checksum = SHA-256 berkas terenkripsi: keduanya diperiksa lab.
# Dua file, dua data key terbungkus berbeda (12 data key per tahun bila satu per bulan).
=> SELECT kunci_objek, length(checksum) AS panjang_checksum, left(kunci_data_terbungkus, 9) AS awal_kunci, versi_kek FROM arsip_partisi ORDER BY periode

$ BAO_TOKEN=<token pembuka> kmsarsip buka -berkas posting_lalu_02.enc -terbungkus <data key terbungkus posting_lalu_01 v1> -periode <tanggal>
