# --- C. Key baru (ledger mewajibkan order_id sama dengan key) dengan akun dan jumlah yang sama adalah permintaan lain; key yang sama dengan jumlah berbeda ditolak ---
$ pembayaran -> ledger  jalur=topup Idempotency-Key=TB-LAB-1003 akun=100001 jumlah=200000
status 201 {"jurnal":<jurnal 2>,"ulang":false}
$ pembayaran -> ledger  jalur=topup Idempotency-Key=TB-LAB-1003 akun=100001 jumlah=300000
status 422 {"alasan":"key-dipakai-permintaan-lain"}
=> SELECT count(DISTINCT j.id) AS jurnal_topup_bank, sum(p.jumlah) FILTER (WHERE p.jumlah > 0) AS total_kredit FROM jurnal j JOIN posting p ON p.jurnal_id = j.id WHERE j.jenis = 'topup_bank'

-------------------+--------------
