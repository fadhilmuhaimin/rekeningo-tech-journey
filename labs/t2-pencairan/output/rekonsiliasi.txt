(1 row)
=> INSERT INTO posting (jurnal_id, buku_id, jumlah) VALUES
     (26, (SELECT id FROM buku WHERE akun_id = 508), 15000),
     (26, (SELECT id FROM buku WHERE akun_id IS NULL AND jenis = 'kas_pencairan'), -15000);
INSERT 0 2
=> UPDATE buku b SET saldo_cache = b.saldo_cache + p.jumlah
   FROM (SELECT buku_id, sum(jumlah) AS jumlah FROM posting WHERE jurnal_id = 26 GROUP BY buku_id) p
   WHERE b.id = p.buku_id;
UPDATE 2
=> COMMIT;
COMMIT
