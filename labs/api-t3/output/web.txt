# --- A. Login dasbor: cookie sesi tidak terbaca JavaScript, token CSRF dikirim di body ---
$ curl -i -X POST '/web/login' -d '{"email": "warung.ani@lestari.example", "password": "sementara-tahap-3"}'
HTTP/1.1 200 OK
Content-Type: application/json
Set-Cookie: __Host-sesi=<token sesi web Ani>; Path=/; Max-Age=28800; HttpOnly; Secure; SameSite=Lax

{"csrf":"<token CSRF Ani>","kedaluwarsa_detik":28800}

# Database menyimpan hash token sesi dan hash token CSRF, bukan tokennya

# --- B. Membaca dengan cookie saja jalan; mengubah data butuh token CSRF ---
$ curl -i -X GET '/warung/418/penerimaan?batas=1' -H 'Cookie: __Host-sesi=<token sesi web Ani>'
HTTP/1.1 200 OK
Content-Type: application/json

{"penerimaan":[],"warung":418}
$ curl -i -X PATCH '/produk/1' -H 'Cookie: __Host-sesi=<token sesi web Ani>' -d '{"stok": 5, "versi": 1}'
HTTP/1.1 403 Forbidden
Content-Type: application/problem+json

{"type":"/problems/csrf","title":"Permintaan ubah dari dasbor ditolak","status":403,"detail":"Kirim token CSRF dari POST /web/login di header X-CSRF-Token, dari halaman yang sama asalnya."}
$ curl -i -X PATCH '/produk/1' -H 'Cookie: __Host-sesi=<token sesi web Ani>' -H 'X-CSRF-Token: salah' -d '{"stok": 5, "versi": 1}'
HTTP/1.1 403 Forbidden
Content-Type: application/problem+json

{"type":"/problems/csrf","title":"Permintaan ubah dari dasbor ditolak","status":403,"detail":"Kirim token CSRF dari POST /web/login di header X-CSRF-Token, dari halaman yang sama asalnya."}
$ curl -i -X PATCH '/produk/1' -H 'Cookie: __Host-sesi=<token sesi web Ani>' -H 'X-CSRF-Token: <token CSRF Ani>' -H 'Sec-Fetch-Site: cross-site' -d '{"stok": 5, "versi": 1}'
HTTP/1.1 403 Forbidden
Content-Type: application/problem+json

{"type":"/problems/csrf","title":"Permintaan ubah dari dasbor ditolak","status":403,"detail":"Kirim token CSRF dari POST /web/login di header X-CSRF-Token, dari halaman yang sama asalnya."}
$ curl -i -X PATCH '/produk/1' -H 'Cookie: __Host-sesi=<token sesi web Ani>' -H 'X-CSRF-Token: <token CSRF Ani>' -H 'Origin: http://situs-lain.example' -d '{"stok": 5, "versi": 1}'
HTTP/1.1 403 Forbidden
Content-Type: application/problem+json

{"type":"/problems/csrf","title":"Permintaan ubah dari dasbor ditolak","status":403,"detail":"Kirim token CSRF dari POST /web/login di header X-CSRF-Token, dari halaman yang sama asalnya."}
$ curl -i -X PATCH '/produk/1' -H 'Cookie: __Host-sesi=<token sesi web Ani>' -H 'X-CSRF-Token: <token CSRF Ani>' -H 'Sec-Fetch-Site: same-origin' -d '{"stok":5,"versi":1}'
HTTP/1.1 200 OK
Content-Type: application/json

{"id":1,"harga":25000,"stok":5,"versi":2}
# Cookie SameSite=Lax saja tidak cukup: server memeriksa Sec-Fetch-Site, Origin, dan token (OWASP CSRF Prevention, R3-19)

# --- C. Token dari satu jalur tidak berlaku di jalur lain; karyawan tidak punya dasbor ---
$ curl -i -X GET '/warung/418/penerimaan?batas=1' -H 'Cookie: __Host-sesi=<token sesi Warung Ani (app)>'
