# --- B. Pembanding: satu baris outbox per penerima (cara lama), ukuran dibaca dari katalog PostgreSQL ---
=> INSERT INTO outbox (akun_id, kelas, jenis, payload, tujuan)
              SELECT akun_id, 2, 'promo', jsonb_build_object('judul', 'Diskon makan siang', 'isi', ''), 'push' FROM perangkat WHERE aktif
INSERT 0 10281
-- baris outbox per penerima: 10281
-- tumpukan tabel (pg_relation_size) bertambah 166 byte per baris, tanpa index (dibulatkan ke bawah)
-- Hitungan: 166 byte x 1.000.000 penerima = 166 MB di tumpukan tabel; naskah memakai 300 byte per baris (Asumsi, dengan index): 300 MB
