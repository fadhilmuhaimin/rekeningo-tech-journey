# Tiap snapshot dibandingkan dengan hitungan langsung atas tabel posting (saldo dan jumlah posting sampai akhir bulannya): 0 baris berbeda dari 113.168.

# --- C. Pemeriksaan malam dari snapshot terakhir: posting sesudahnya saja ---
$ ./api-t5 -cek-cache -dari-snapshot
cek cache dari snapshot <tanggal>: 10288 buku diperiksa, 0 tidak cocok
# exit status 0
# Baris posting dibaca: 150.000 dari 1.800.000, yaitu 8,3% (satu bulan dari 12).
# Snapshot yang lebih tua membuat pemeriksaan membaca lebih banyak bulan: -snapshot-periode memilih periodenya.
$ ./api-t5 -cek-cache -dari-snapshot -snapshot-periode <tanggal 1 tiga bulan lalu>
cek cache dari snapshot <tanggal>: 10288 buku diperiksa, 0 tidak cocok
# exit status 0
# Baris posting dibaca: 450.000, yaitu tiga bulan (dua bulan lalu, bulan lalu, dan bulan berjalan) × 150.000 posting.

# --- D. Jawaban yang sama: satu selisih cache yang nyata terlihat oleh kedua pemeriksaan ---
# Kode yang menulis saldo_cache tanpa posting dibuat dengan UPDATE sebagai role aplikasi (api_t5 memegang hak UPDATE atas kolom itu).
(api_t5) => UPDATE buku SET saldo_cache = saldo_cache + 5000 WHERE id = (SELECT min(id) FROM buku WHERE jenis = 'karyawan')
UPDATE 1
$ ./api-t5 -cek-cache
buku 8 (karyawan Dimas): saldo_cache 431000, jumlah posting 426000, selisih 5000
cek cache: 10288 buku diperiksa, 1 tidak cocok
# exit status 1
$ ./api-t5 -cek-cache -dari-snapshot
buku 8 (karyawan Dimas): saldo_cache 431000, snapshot + posting sesudahnya 426000, selisih 5000
cek cache dari snapshot <tanggal>: 10288 buku diperiksa, 1 tidak cocok
# exit status 1
# Kedua pemeriksaan menandai satu buku yang sama dengan saldo_cache, hitungan, dan selisih yang sama (5000).
(api_t5) => UPDATE buku SET saldo_cache = saldo_cache - 5000 WHERE id = (SELECT min(id) FROM buku WHERE jenis = 'karyawan')
UPDATE 1
