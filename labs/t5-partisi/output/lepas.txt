# --- D. Membuang satu bulan: DETACH dan DROP lawan DELETE, dengan jumlah baris sama (tabel uji tanpa trigger ledger) ---
# DELETE atas posting ditolak ledger (bagian B), jadi pembandingnya adalah dua tabel uji dengan kolom dan index yang sama dengan posting,
# masing-masing 200.000 baris di bulan yang dibuang dan 200.000 baris di bulan lain. Waktunya hanya diperiksa terhadap batas.
=> CREATE TABLE uji_biasa (LIKE posting INCLUDING DEFAULTS INCLUDING INDEXES)

=> CREATE TABLE uji_part (LIKE posting INCLUDING DEFAULTS INCLUDING INDEXES) PARTITION BY RANGE (dibuat)
CREATE TABLE
=> CREATE TABLE uji_part_a PARTITION OF uji_part FOR VALUES FROM ('<waktu>') TO ('<waktu>')
CREATE TABLE
=> CREATE TABLE uji_part_b PARTITION OF uji_part FOR VALUES FROM ('<waktu>') TO ('<waktu>')
CREATE TABLE
=> SELECT (SELECT count(*) FROM uji_biasa) AS uji_biasa, (SELECT count(*) FROM uji_part) AS uji_part, (SELECT count(*) FROM uji_part_a) AS bulan_dibuang

-----------+----------+---------------
 400000    | 400000   | 200000
(1 row)
=> DELETE FROM uji_biasa WHERE <bulan yang dibuang>
DELETE 200000
=> ALTER TABLE uji_part DETACH PARTITION uji_part_a CONCURRENTLY; DROP TABLE uji_part_a
DROP TABLE   -- 200000 baris di partisi yang dibuang
# DELETE dan DETACH + DROP membuang 200.000 baris yang sama. Batas yang diperiksa: DELETE memerlukan lebih dari dua kali waktu DETACH + DROP. Nilai waktunya tidak direkam.
=> SELECT (SELECT count(*) FROM uji_biasa) AS uji_biasa, (SELECT count(*) FROM uji_part) AS uji_part

-----------+----------
 200000    | 200000
(1 row)
=> DROP TABLE uji_biasa, uji_part
