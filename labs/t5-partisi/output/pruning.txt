# Mencari satu baris dengan id saja membaca 14 dari 14 partisi: index primary key di 12 partisi berisi dan Seq Scan di 2 partisi kosong.
# id tidak memuat kolom kunci partisi, jadi batas partisi tidak bisa menyisihkan satu pun.

# --- B2. Parallel Append tidak menunggu partisi sebesar 8 MB ---
# min_parallel_table_scan_size (bawaan 8 MB) menentukan apakah tabel biasa boleh dipindai paralel. Kode planner mengecualikan partisi
# (anak inheritance) dari batas itu (PostgreSQL 17, allpaths.c, compute_parallel_worker). Diukur di sini: 12 partisi bulanan berisi 30.000 baris.
=> CREATE TABLE uji_kecil (LIKE posting INCLUDING DEFAULTS) PARTITION BY RANGE (dibuat)
CREATE TABLE
# 12 partisi bulanan uji_kecil_NN dibuat dengan batas yang sama dengan posting, masing-masing diisi 30.000 baris.
# Partisi terbesar kurang dari 8 MB. Plan untuk query tanpa dibuat: Parallel Append dengan 12 Parallel Seq Scan, satu per partisi.
# Jadi ukuran partisi bukan syarat Parallel Append. Seberapa kecil partisi boleh sebelum planner memilih Append biasa tidak diukur di sini.
=> DROP TABLE uji_kecil
