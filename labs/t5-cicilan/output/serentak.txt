# --- A. Dua job berjalan serentak atas 24 angsuran jatuh tempo (enam jadwal, versi benar) ---
# enam karyawan, tiap jadwal empat angsuran Rp10.000 ke Warung Ani; keempatnya sudah jatuh tempo (jadwal mulai tiga bulan lalu)
=> SELECT status, count(*) AS angsuran FROM angsuran WHERE jadwal_id = ANY (ARRAY[1, 2, 3, 4, 5, 6]) GROUP BY status ORDER BY status
 status   | angsuran
----------+----------
 menunggu | 24
(1 row)
# dua proses job dimulai bersamaan; tumpang tindihnya tidak dipaksa (bagian B memaksanya). Yang direkam jumlah gabungan kedua job, karena pembagian kerja antar proses bergantung waktu
jurnal baru dari kedua job: 24; angsuran dicatat dari kedua job: 24
=> SELECT count(*) AS angsuran, count(*) FILTER (WHERE status = 'dicatat') AS dicatat,
                      (SELECT count(*) FROM jurnal WHERE jenis = 'angsuran' AND referensi LIKE 'angsuran:%') AS jurnal_angsuran,
                      (SELECT count(DISTINCT referensi) FROM jurnal WHERE jenis = 'angsuran') AS referensi_berbeda
               FROM angsuran WHERE jadwal_id = ANY (ARRAY[1, 2, 3, 4, 5, 6])
 angsuran | dicatat | jurnal_angsuran | referensi_berbeda
----------+---------+-----------------+-------------------
 24       | 24      | 24              | 24
(1 row)
=> SELECT DISTINCT 250000 - (SELECT sum(p.jumlah) FROM posting p JOIN buku b ON b.id = p.buku_id WHERE b.akun_id = j.akun_id AND b.jenis = 'karyawan') AS berkurang
               FROM jadwal_cicilan j WHERE j.id = ANY (ARRAY[1, 2, 3, 4, 5, 6])
 berkurang
-----------
 40000
(1 row)

# --- B. Dua job sudah membaca angsuran yang sama, lalu berebut buku; saldo hanya cukup untuk satu angsuran ---
# akun 10007 bersaldo Rp250.000; jadwal 7: dua angsuran sebesar saldo itu, hanya yang ke-1 jatuh tempo
=> SELECT ke, status FROM angsuran WHERE jadwal_id = 7 ORDER BY ke
 ke | status
----+----------
 1  | menunggu
 2  | menunggu
(2 rows)
# pemilik menahan lock buku kedua akun; dua job dimulai dan dibiarkan menunggu di lock
koneksi job yang menunggu lock: 2
# lock dilepas; jumlah gabungan kedua job:
jurnal baru 1, jurnal sudah ada 1, gagal 0
=> SELECT ke, status, jurnal_id IS NOT NULL AS terikat_jurnal FROM angsuran WHERE jadwal_id = 7 ORDER BY ke
 ke | status   | terikat_jurnal
----+----------+----------------
 1  | dicatat  | t
 2  | menunggu | f
(2 rows)
=> SELECT count(*) AS jurnal_angsuran_ke_1 FROM jurnal WHERE referensi = 'angsuran:7:1'
 jurnal_angsuran_ke_1
----------------------
 1
(1 row)
# saldo akun 10007 dari jumlah posting: Rp0
