# --- A. Angsuran sudah dijurnal, langkah akhirnya tidak bisa selesai: jurnal pembalik dan status gagal ---
# Lab ini tidak punya langkah akhir yang bisa gagal; yang direkam adalah mekanisme kompensasinya. Keadaan 'terjurnal' dipasang langsung di database.
=> UPDATE angsuran SET status = 'terjurnal', jurnal_id = (SELECT id FROM jurnal WHERE referensi = 'angsuran:1:1')
               WHERE jadwal_id = 1 AND ke = 1
UPDATE 1
=> SELECT ke, status, jurnal_id IS NOT NULL AS terikat_jurnal FROM angsuran WHERE jadwal_id = 1 AND ke = 1

=> SELECT j.jenis, (j.membalik_jurnal_id = a.id) AS membalik_jurnal_angsuran
