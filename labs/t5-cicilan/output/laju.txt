# --- A. Job satu proses atas 2.000 angsuran jatuh tempo (200 jadwal x 10 angsuran Rp1.000, 20 warung), versi benar ---
# Jadwal ditulis langsung di database (bukan lewat POST /cicilan) supaya 2.000 angsuran siap dalam satu perintah; job-nya job yang sama.
=> WITH w AS (SELECT id, row_number() OVER (ORDER BY id) AS n FROM akun WHERE jenis = 'warung' ORDER BY id LIMIT 20),
                   k AS (SELECT id, row_number() OVER (ORDER BY id) AS n FROM akun WHERE jenis = 'karyawan' ORDER BY id OFFSET 3 LIMIT 200)
              INSERT INTO jadwal_cicilan (akun_id, jenis, merchant_akun_id, jumlah_per_angsuran, jumlah_angsuran, hari_tagih)
              SELECT k.id, 'cicilan', w.id, 1000, 10, 15 FROM k JOIN w ON w.n = (k.n - 1) % 20 + 1 ORDER BY k.id
INSERT 0 200
=> INSERT INTO angsuran (jadwal_id, ke, jatuh_tempo) SELECT j.id, k, current_date - 1 FROM jadwal_cicilan j, generate_series(1, 10) k
