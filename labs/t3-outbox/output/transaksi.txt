# --- B. Sebelum COMMIT worker tidak melihat baris outbox; sesudah COMMIT terlihat dan terkirim ---
# Sesi tangan (pemilik schema): jurnal 90001 seimbang (-1000 dan +1000) dan satu baris outbox, belum COMMIT
=> BEGIN
=> INSERT INTO jurnal (id, jenis, keterangan, dibuat_oleh) VALUES (90001, 'bayar', 'sesi tangan', 419)
INSERT 0 1
=> INSERT INTO posting (jurnal_id, buku_id, jumlah) SELECT 90001, id, -1000 FROM buku WHERE akun_id = 419
INSERT 0 1
=> INSERT INTO posting (jurnal_id, buku_id, jumlah) SELECT 90001, id, 1000 FROM buku WHERE akun_id = 418
INSERT 0 1
=> INSERT INTO outbox (jurnal_id, akun_id, kelas, jenis, payload) VALUES (90001, 418, 0, 'bayar', '{"judul": "sesi tangan"}')
INSERT 0 1
=> SELECT count(*) FROM outbox WHERE jurnal_id = 90001   # sesi tangan sendiri: 1; sesi lain (worker termasuk): 0
$ ./api-t3 -peran worker -sekali
worker: diklaim 0, terkirim 0, dilewati 0, ditunda 0, mati 0
# Worker berjalan sementara transaction terbuka: diklaim 0. Pemberitahuan tidak bisa mendahului uangnya.
=> COMMIT
$ ./api-t3 -peran worker -sekali
worker: diklaim 1, terkirim 1, dilewati 0, ditunda 0, mati 0

# --- C. ROLLBACK membuang jurnal dan baris outbox bersama-sama ---
=> BEGIN; jurnal 90002 seimbang, posting, dan satu baris outbox seperti di B
=> ROLLBACK
=> SELECT (SELECT count(*) FROM jurnal WHERE id = 90002) AS jurnal_90002, (SELECT count(*) FROM outbox WHERE jurnal_id = 90002) AS outbox_90002

# --- D. Pembayaran ditolak (saldo kurang) dan COMMIT ditolak database: tidak ada jurnal, tidak ada baris outbox ---
