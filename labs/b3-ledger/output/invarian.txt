# --- A. Invarian 1: jurnal harus seimbang (diperiksa saat COMMIT) ---
-- Top-up dengan salah ketik: tunjangan -250.000, Budi +25.000
=> BEGIN;
BEGIN
=> INSERT INTO jurnal (jenis, keterangan, dibuat_oleh) VALUES ('topup', 'Tunjangan makan, salah ketik', 400) RETURNING id;
 id
----
 12
(1 row)
=> INSERT INTO posting (jurnal_id, buku_id, jumlah) VALUES (12, 1, -250000), (12, 7, 25000);
INSERT 0 2
=> UPDATE buku b SET saldo_cache = b.saldo_cache + p.jumlah
   FROM (SELECT buku_id, sum(jumlah) AS jumlah FROM posting WHERE jurnal_id = 12 GROUP BY buku_id) p
   WHERE b.id = p.buku_id;
UPDATE 2
=> COMMIT;
ERROR:  jurnal 12 tidak seimbang: 2 posting, jumlah -225000
HINT:  Jurnal butuh minimal dua posting yang jumlahnya 0.
   -- SQLSTATE 23514 check_violation, constraint jurnal_seimbang
=> SELECT count(*) AS jurnal_12 FROM jurnal WHERE id = 12;
 jurnal_12
-----------
         0
(1 row)
