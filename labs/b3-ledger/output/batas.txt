# --- 2. Tanpa lock, pemeriksaan A dan B selesai sebelum salah satunya COMMIT ---
-- SET CONSTRAINTS ALL IMMEDIATE menjalankan pemeriksaan sekarang, bukan saat COMMIT, supaya urutannya bisa
-- diulang. Tanpa perintah ini pemeriksaan berjalan di dalam COMMIT, sebelum hasilnya terlihat transaction
-- lain, jadi celahnya tetap ada (2b). Pemeriksaan B tidak melihat posting A yang belum COMMIT (READ COMMITTED).
[A] => BEGIN;
BEGIN
[A] => SELECT sum(jumlah) AS saldo_budi FROM posting WHERE buku_id = 2;
 saldo_budi
------------
     100000
(1 row)

  2
(1 row)
[A] => INSERT INTO posting (jurnal_id, buku_id, jumlah) VALUES (2, 2, -70000), (2, 3, 70000);
INSERT 0 2
[A] => SET CONSTRAINTS ALL IMMEDIATE;
SET CONSTRAINTS
[B] => BEGIN;
BEGIN
[B] => SELECT sum(jumlah) AS saldo_budi FROM posting WHERE buku_id = 2;
 saldo_budi
------------
     100000
(1 row)
[B] => INSERT INTO jurnal (jenis, keterangan, dibuat_oleh) VALUES ('transfer', 'Transfer ke Dimas', 419) RETURNING id;

SET CONSTRAINTS
[A] => COMMIT;
COMMIT

=> SELECT (SELECT sum(jumlah) FROM posting WHERE buku_id = 2) AS saldo_budi,

(1 row)
-- Dua-duanya COMMIT. Saldo Budi -Rp20.000: invarian 3 tanpa lock hanya benar untuk satu transaction.

# --- 3. saldo_cache diperbarui di transaction yang sama: UPDATE mengunci baris buku Budi ---
-- Pola kejadian.txt. B menunggu lock baris buku sampai A COMMIT, lalu pemeriksaan B melihat jurnal A.
[A] => BEGIN;
BEGIN
[A] => SELECT sum(jumlah) AS saldo_budi FROM posting WHERE buku_id = 2;
 saldo_budi
------------
     100000
(1 row)
[A] => INSERT INTO jurnal (jenis, keterangan, dibuat_oleh) VALUES ('transfer', 'Transfer ke Citra', 419) RETURNING id;
 id
----
  2
(1 row)
[A] => INSERT INTO posting (jurnal_id, buku_id, jumlah) VALUES (2, 2, -70000), (2, 3, 70000);
INSERT 0 2
[A] => UPDATE buku b SET saldo_cache = b.saldo_cache + p.jumlah
       FROM (SELECT buku_id, sum(jumlah) AS jumlah FROM posting WHERE jurnal_id = 2 GROUP BY buku_id) p
       WHERE b.id = p.buku_id;
UPDATE 2
[B] => BEGIN;
BEGIN
[B] => SELECT sum(jumlah) AS saldo_budi FROM posting WHERE buku_id = 2;
 saldo_budi

[B] => UPDATE buku b SET saldo_cache = b.saldo_cache + p.jumlah
       FROM (SELECT buku_id, sum(jumlah) AS jumlah FROM posting WHERE jurnal_id = 3 GROUP BY buku_id) p
       WHERE b.id = p.buku_id;
