# --- B. v1: kenapa saldo Budi Rp180.000? Tiga tabel digabung dengan tangan ---
=> SELECT 'topup' AS dari_tabel, nominal AS jumlah, keterangan FROM topup WHERE akun_id = 419
   UNION ALL
   SELECT 'transaksi', -jumlah, 'bayar ke akun ' || ke FROM transaksi WHERE dari = 419
   UNION ALL
   SELECT 'transaksi', jumlah, 'diterima dari akun ' || dari FROM transaksi WHERE ke = 419
   UNION ALL
   SELECT 'koreksi', jumlah, alasan FROM koreksi WHERE akun_id = 419
   ORDER BY dari_tabel DESC;
 dari_tabel | jumlah |                                                       keterangan
------------+--------+------------------------------------------------------------------------------------------------------------------------
 transaksi  | -70000 | bayar ke akun 418
 topup      | 250000 | Tunjangan makan minggu 1
 koreksi    |  70000 | Pembayaran Budi ke Warung Ani, Jumat minggu 3 pukul 12.15, gagal di tengah; saldo terpotong tanpa baris transaksi (M4)
(3 rows)
=> SELECT saldo AS kolom_saldo,
          (SELECT coalesce(sum(nominal), 0) FROM topup WHERE akun_id = 419)
        - (SELECT coalesce(sum(jumlah), 0) FROM transaksi WHERE dari = 419)
        + (SELECT coalesce(sum(jumlah), 0) FROM transaksi WHERE ke = 419)
        + (SELECT coalesce(sum(jumlah), 0) FROM koreksi WHERE akun_id = 419) AS dari_tiga_tabel
   FROM akun WHERE id = 419;
 kolom_saldo | dari_tiga_tabel
-------------+-----------------
      180000 |          250000
(1 row)
-- Tiga tabel menjelaskan Rp250.000, kolom saldo menulis Rp180.000. Rp70.000 yang terpotong saat M4
-- tidak punya baris; yang tercatat hanya koreksinya.

# --- E. Saldo tiap akun sebelum dan sesudah migrasi ---
=> SELECT v1.id AS akun, v1.nama, v1.saldo AS saldo_v1, b.id AS buku,
          coalesce(sum(p.jumlah), 0) AS jumlah_posting, b.saldo_cache,
          v1.saldo = coalesce(sum(p.jumlah), 0) AND v1.saldo = coalesce(b.saldo_cache, 0) AS sama
   FROM b3_ledger_v1.akun v1
   LEFT JOIN buku b ON b.akun_id = v1.id
   LEFT JOIN posting p ON p.buku_id = b.id
   GROUP BY v1.id, v1.nama, v1.saldo, b.id
   ORDER BY v1.id;
