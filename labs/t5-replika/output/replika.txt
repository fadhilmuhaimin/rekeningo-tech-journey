# --- B. Standby menolak DML dan DDL ---
# Pernyataan yang mengubah data, skema, atau mengunci baris ditolak dengan kode 25006 di standby. Pesan galatnya milik mode
# baca-saja standby, bukan milik trigger append-only ledger.
replika=> UPDATE t5_ledger.buku SET saldo_cache = saldo_cache + 1 WHERE id = 10
