# --- A. API ledger: satu jurnal per Idempotency-Key (permintaan langsung ke ledger, tanpa gateway) ---
# Akun Budi (419) punya saldo awal dari data awal; top-up Rp200.000 dengan key = order_id.
# saldo Budi sebelum: 250000
# tanpa kunci layanan
$ curl -i -X POST '/ledger/jurnal-topup' -H 'Idempotency-Key: <order A>' -d '{"order_id": "<order A>", "akun_id": 419, "jumlah": 200000}'
HTTP/1.1 401 Unauthorized
Content-Type: application/json

{"alasan":"kunci-layanan"}
# tanpa Idempotency-Key
$ curl -i -X POST '/ledger/jurnal-topup' -H 'X-Layanan-Kunci: kunci-layanan-pembayaran-lab' -d '{"order_id": "<order A>", "akun_id": 419, "jumlah": 200000}'
HTTP/1.1 400 Bad Request
Content-Type: application/json

{"alasan":"idempotency-key-wajib"}
# Idempotency-Key bukan order_id
$ curl -i -X POST '/ledger/jurnal-topup' -H 'X-Layanan-Kunci: kunci-layanan-pembayaran-lab' -H 'Idempotency-Key: lain' -d '{"order_id": "<order A>", "akun_id": 419, "jumlah": 200000}'
HTTP/1.1 422 Unprocessable Entity
Content-Type: application/json

{"alasan":"tidak-sah","field":"Idempotency-Key"}
# permintaan sah, lalu ulangan yang sama
$ curl -i -X POST '/ledger/jurnal-topup' -H 'X-Layanan-Kunci: kunci-layanan-pembayaran-lab' -H 'Idempotency-Key: <order A>' -d '{"order_id": "<order A>", "akun_id": 419, "jumlah": 200000}'

# --- B. Webhook dari gateway lewat service pembayaran: kiriman yang sama, dan event berbeda untuk kejadian yang sama ---
# proses: gateway tiruan, monolith (role t4m_monolit, API ledger), service pembayaran (role t4m_pembayaran)
# kiriman pertama event e1
hasil: 200 diproses
# kiriman e1 yang sama lagi, lalu event e2 (berbeda) untuk kejadian yang sama
hasil: 200 ganda; 200 ganda
# keadaan order di tiga modul
=> SELECT t.status AS topup_bank, t.jurnal_id IS NOT NULL AS terikat_jurnal,
                        (SELECT count(*) FROM t4m_ledger.jurnal WHERE jenis = 'topup_bank' AND keterangan = 'Top-up bank ' || t.order_id) AS jurnal_di_ledger,
                        (SELECT count(*) FROM t4m_platform.outbox o JOIN t4m_ledger.jurnal j ON j.id = o.jurnal_id WHERE j.keterangan = 'Top-up bank ' || t.order_id) AS outbox_di_platform
                 FROM t4m_pembayaran.topup_bank t WHERE t.order_id = '<order 1>'
 topup_bank | terikat_jurnal | jurnal_di_ledger | outbox_di_platform
------------+----------------+------------------+--------------------
 dibayar    | t              | 1                | 1
(1 row)
=> SELECT w.event_id, w.hasil, w.jurnal_id IS NOT NULL AS ada_jurnal FROM t4m_pembayaran.webhook_event w JOIN t4m_pembayaran.topup_bank t ON t.id = w.topup_id WHERE t.order_id = '<order 1>' ORDER BY w.event_id
 event_id | hasil    | ada_jurnal
----------+----------+------------
 e1       | diproses | t
 e2       | ganda    | t
(2 rows)
